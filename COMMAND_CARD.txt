┌──────────────────────────────────────────────────────────────────────┐
│ 🔨 Complete Rebuild (Nuclear Option)                                 │
└──────────────────────────────────────────────────────────────────────┘

cd /root/uvis

# 1. Stop everything
docker-compose down

# 2. Remove images
docker rmi uvis-frontend 2>/dev/null || true

# 3. Clean local build
rm -rf frontend/dist/ frontend/node_modules/.vite

# 4. Fresh build
cd frontend && npm run build && cd ..

# 5. Verify local build
ls -lh frontend/dist/index.html
ls frontend/dist/assets/*.js | wc -l

# 6. Docker rebuild (no cache)
docker-compose build --no-cache frontend

# 7. Start
docker-compose up -d

# 8. Wait
sleep 30

# 9. Check deployment
docker exec uvis-frontend sh -c "ls /usr/share/nginx/html/assets/*.js | wc -l"
